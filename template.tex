  \begin{posterbullets}
    \item Compress the method into three steps: input, transformation, output.
    \item Keep equations only if they are central to the contribution.
    \item Use labels that match the hero visual.
  \end{posterbullets}

  \begin{posterbullets}

  \end{posterbullets}
