\renewcommand{\posterstatementcard}[5]{%
  \begingroup\setlength{\fboxsep}{0.42em}%
  \colorbox{sidebarcool}{%
    \begin{minipage}{0.94\linewidth}%
      {\fontsize{#1}{#2}\selectfont #3\textcolor{black}{#4}\par}%
      \vspace{0.14em}{\fontsize{35}{45}\selectfont\color{black}#5\par}%
    \end{minipage}}%
  \endgroup\par\vspace{0.45em}%
}

}

  \theoremcard{Theorem}{Assumptions, formal statement, and main implication.}
  \propositioncard{Proposition}{Supporting result, technical condition, and interpretation.}

  Contribution summary, broader implication, and limitation.
