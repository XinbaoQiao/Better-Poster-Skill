  \section{Theory Claims}
  \propositioncard{Proposition 1: Replace paradigm}{%
    Under the \textit{Replace} paradigm, as $t\to\infty$,
    \[
      \boldsymbol{\Sigma}_{t}\xrightarrow{a.s.}\mathbf{0},
      \qquad
      \mathbb{E}\!\left[\mathbb{W}_{2}^{2}(\mathcal{N}_{t},\mathcal{N}_{0})\right]\to\infty.
    \]}

  \propositioncard{Proposition 2: Accumulate paradigm}{%
    Under unbiased accumulation,
    \[
      \mathbb{E}[(\bar{\boldsymbol{\mu}}_{t}-\bar{\boldsymbol{\mu}}_{0})^{2}]
      \to (1-\alpha_{n})\bar{\boldsymbol{\Sigma}}_{0},
      \quad
      \mathbb{E}[\bar{\boldsymbol{\Sigma}}_{t}]
      \to \alpha_{n}\bar{\boldsymbol{\Sigma}}_{0}.
    \]
    Thus $\mathbb{E}[\mathbb{W}_{2}^{2}]
    \to 2(1-\sqrt{\alpha_{n}})\operatorname{Tr}(\bar{\boldsymbol{\Sigma}}_{0})$.}

  \theoremcard{Theorem 1: Selection bias precipitates collapse}{%
    With top-$\alpha$ selection toward a local ideal $\mathbf{u}^{*}$,
    \[
      \|\bar{\boldsymbol{\mu}}_{t}-\mathbf{u}^{*}\|^{2}\!\to 0,
      \qquad
      \bar{\boldsymbol{\Sigma}}_{t}\!\to \mathbf{0}.
    \]
    Local fidelity can erase global diversity.}

  \theoremcard{Theorem 2: Collapse has an explicit rate}{%
    If the trajectory remains in the local basin,
    \[
      {\rm Tr}(\bar{\boldsymbol{\Sigma}}_{t})
      = \mathcal{O}_{a.s.}(t^{-\psi}).
    \]
    The tail erosion follows a power-law decay.}

  \theoremcard{Theorem 3: Collapse costs true-manifold risk}{%
    For filtered distribution $\mathcal{D}_{t}$ and true manifold
    $\mathcal{D}^{*}$,
    \[
      \mathcal{R}_{\mathcal{D}^{*}}(h_{t};g^{*})
      \le 2\ell\epsilon\,\mathbb{W}_{p}(\mathcal{D}_{t},\mathcal{D}^{*})
      +\mathcal{R}_{\mathcal{D}_{t}}(h_{t};g_{t})
      +\mathcal{O}(\ell\delta).
    \]}

  \section{Proxy Principle}
  A collaborative Wasserstein barycenter broadens the verifier:
  \[
    \mathcal{Q}^{*}=\arg\min_{\mathcal{Q}}
    \sum_{k=1}^{K}\lambda_{k}\mathbb{W}_{p}^{p}(\mathcal{Q},\mathcal{Q}_{k}).
  \]
  Selection is protective only when this proxy covers the modes the loop
  must preserve.
