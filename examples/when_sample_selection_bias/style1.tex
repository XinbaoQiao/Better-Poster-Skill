  \section{Introduction}
  Recursive synthetic-data training can narrow distributional support when later generations are selected by biased local references. The main question is when accumulation preserves diversity, and when selection bias overrides it.

  \section{Theory}
  \propositioncard{Replace paradigm [R1]}{Replacing data by the latest generated round drives covariance loss and mode narrowing.}
  \propositioncard{Accumulate paradigm [R2]}{Pooling generations can stabilize unbiased recursion, but biased selection changes the limiting distribution.}
  \theoremcard{Theorem 1}{Top-rank selection toward a local reference moves the mean toward that reference while covariance contracts.}
  \theoremcard{Theorem 2}{Within the local basin, the covariance trace follows a power-law decay, giving an explicit collapse rate.}
  \theoremcard{Theorem 3}{After fitting error is small, target performance is governed by distributional mismatch.}

  \section{Multivariate Gaussian Modeling}
  \propositioncard{Gaussian}{Selection contracts covariance under both replacement and accumulation.}
  \propositioncard{Anisotropic Gaussian}{Directional variance changes the path, but not the loss of dispersion.}
  \propositioncard{Gaussian mixture}{Biased filtering reduces mode coverage across generations.}
  \propositioncard{Laplace}{Heavy-tailed initialization still loses dispersion under biased filtering.}

  \section{Collapse Signal}
  \theoremcard{Trace-ratio diagnostic}{The trace ratio summarizes diversity contraction: replacement changes fastest, while accumulation decays more slowly.}

  \section{Empirical Summary}
  Across Gaussian and non-Gaussian families, biased selection is the dominant driver of diversity loss and reduced coverage.
