  \section{Introduction}
  Synthetic data now flows back into future training sets, so recursive
  training can amplify small distributional errors into model collapse
  (Shumailov et al., Nature 2024). Verification and sample selection are meant
  to stop this loop by keeping samples that match a trusted reference. In
  low-resource silos, however, that reference is local and incomplete: valid
  rare modes can look like bad samples and be filtered away.

  \section{Theory}
  \propositioncard{Proposition 1: Replace paradigm (Shumailov et al., Nature 2024)}{%
    Under the \textit{Replace} paradigm, as $t\to\infty$,
    \[
      \boldsymbol{\Sigma}_{t}\xrightarrow{a.s.}\mathbf{0},
      \qquad
      \mathbb{E}\!\left[\mathbb{W}_{2}^{2}(\mathcal{N}_{t},\mathcal{N}_{0})\right]\to\infty.
    \]}
  \posterstatementsummary{Replacing data eventually removes variance and drifts away.}

  \propositioncard{Proposition 2: Accumulate paradigm (Kazdan et al., ICML 2025)}{%
    Under unbiased accumulation,
    \[
      \mathbb{E}[\bar{\boldsymbol{\Sigma}}_{t}]
      \to \alpha_{n}\bar{\boldsymbol{\Sigma}}_{0}.
    \]
    Thus $\mathbb{E}[\mathbb{W}_{2}^{2}]
    \to 2(1-\sqrt{\alpha_{n}})\operatorname{Tr}(\bar{\boldsymbol{\Sigma}}_{0})$.}
  \posterstatementsummary{Unbiased accumulation keeps diversity from fully collapsing.}

  \theoremcard{Theorem 1: local selection collapses diversity}{%
    With top-$\alpha$ selection toward a local ideal $\mathbf{u}^{*}$,
    \[
      \|\bar{\boldsymbol{\mu}}_{t}-\mathbf{u}^{*}\|^{2}\!\to 0,
      \qquad
      \bar{\boldsymbol{\Sigma}}_{t}\!\to \mathbf{0}.
    \]
    Local fidelity erases global coverage.}
  \posterstatementsummary{Local selection gains fit by erasing global coverage.}

  \theoremcard{Theorem 2: diversity decays by power law}{%
    If the trajectory remains in the local basin,
    \[
      {\rm Tr}(\bar{\boldsymbol{\Sigma}}_{t})
      = \mathcal{O}_{a.s.}(t^{-\psi}).
    \]
    Tail erosion has an explicit rate.}
  \posterstatementsummary{Once trapped locally, diversity shrinks at a predictable rate.}

  \theoremcard{Theorem 3: drift controls true-manifold risk}{%
    For filtered distribution $\mathcal{D}_{t}$ and true manifold
    $\mathcal{D}^{*}$,
    \[
      \mathcal{R}_{\mathcal{D}^{*}}(h_{t};g^{*})
      \le 2\ell\epsilon\,\mathbb{W}_{p}(\mathcal{D}_{t},\mathcal{D}^{*})
      +\mathcal{R}_{\mathcal{D}_{t}}(h_{t};g_{t})
      +\mathcal{O}(\ell\delta).
    \]}
  \posterstatementsummary{Distribution drift upper-bounds risk on true data.}

    other classes.}

  \section{Experimental Takeaway}
  The shown experiments agree: biased selection contracts synthetic
  distributions, and Airplane-only verification makes CIFAR-10 concentrate
  on Airplane. An unshown DDPM benchmark adds the practical check: under
  non-IID references, common selection baselines can lag Random, while
  broader references improve FID and recall.
