  \propositioncard{Proposition: proxy coverage controls drift}{%
    If the verifier proxy covers every relevant mode, the accepted sample
    distribution remains close to the target:
    \[
      W_{2}(\widehat{\mathcal{D}}_{t},\mathcal{D}^{*})
      \le \epsilon_{\rm proxy}+\epsilon_{\rm sample}.
    \]}

  \theoremcard{Theorem: collapse is avoided under coverage}{%
    Under bounded replacement and nonzero mode recall,
    \[
      \Pr[m\in{\rm supp}(\widehat{\mathcal{D}}_{t})]
      \ge 1-\delta,\quad t\le T.
    \]
    The conclusion drives the center claim.}
